<script>
    let currentSpread = 1;
    const totalSpreads = 3;

    function openBook() {
        const cover = document.getElementById('cover');
        cover.style.transform = 'rotateY(-135deg)';
        setTimeout(() => {
            cover.style.display = 'none';
            document.getElementById('nav').style.display = 'flex';
        }, 800);
    }

    function updatePages() {
        for (let i = 1; i <= 5; i++) {
            const page = document.getElementById('spread' + i);
            if (page) page.classList.remove('active');
        }

        if (currentSpread === 1) {
            document.getElementById('spread1').classList.add('active');
            document.getElementById('spread2').classList.add('active');
        } else if (currentSpread === 2) {
            document.getElementById('spread3').classList.add('active');
            document.getElementById('spread4').classList.add('active');
        } else if (currentSpread === 3) {
            document.getElementById('spread5').classList.add('active');
        }
    }

    function nextPage() {
        if (currentSpread < totalSpreads) {
            currentSpread++;
            updatePages();
        }
    }

    function prevPage() {
        if (currentSpread > 1) {
            currentSpread--;
            updatePages();
        }
    }

    updatePages();
</script>
